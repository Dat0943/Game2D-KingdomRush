void CheckForTable()
{
    if (!playerMovement.canSit) return;

    RaycastHit hit;
    if (Physics.Raycast(cameraTransform.position - new Vector3(0f, offset, 0f), cameraTransform.forward, out hit, detectionRange, LayerConsts.OBSTACLE_LAYER))
    {
        Chair chair = hit.collider.GetComponentInParent<Chair>();

        if (chair != null)
        {
            if (currentTargetChair != chair)
            {
                currentTargetChair = chair;
                currentTargetTable = chair.myTable;
                currentTargetTable.currentTargetChair = currentTargetChair;
                currentTargetTable.playerMov = playerMovement;
                SetInteractionState(true);
            }
            return;
        }
    }
    
    if (currentTargetChair != null || currentTargetTable != null)
    {
        SetInteractionState(false);
        currentTargetTable.currentTargetChair = null;
        currentTargetTable.playerMov = null;
        currentTargetChair = null;
        currentTargetTable = null;
    }
}

void SetInteractionState(bool isLookingAtTable)
{
    if (currentTargetTable == null) return;
    if (currentTargetChair.isSitted) return;

    if (currentTargetTable.joinButton != null)
        currentTargetTable.ActiveJoinButton(isLookingAtTable);

    if (GuiManager.Ins.touchZone != null)
        GuiManager.Ins.touchZone.GetComponent<Image>().raycastTarget = !isLookingAtTable;

    if(isLookingAtTable)
    {
        if (currentTargetChair.chairtype == Chairtype.RightChair)
            currentTargetTable.SetTransformJoinButton(new Vector3(0f, 25f, -120f));
        else
            currentTargetTable.SetTransformJoinButton(new Vector3(0f, 25f, 120f));
    }
}
